{{- /* Converts a page's rendered HTML to plain Markdown-ish text for llms-full.txt. */ -}} {{- $s := replace .Content " " "" -}} {{- $s = replaceRE `(?is)<(script|style|svg|noscript)\b.*?` "" $s -}} {{- $s = replaceRE `(?is)` "" $s -}} {{- $s = replaceRE `(?i)]*>` "\n\n# " $s -}} {{- $s = replaceRE `(?i)]*>` "\n\n## " $s -}} {{- $s = replaceRE `(?i)]*>` "\n\n### " $s -}} {{- $s = replaceRE `(?i)]*>` "\n\n#### " $s -}} {{- $s = replaceRE `(?i)` "\n\n" $s -}} {{- $s = replaceRE `(?i)]*>` "\n- " $s -}} {{- $s = replaceRE `(?i)` "**" $s -}} {{- $s = replaceRE `(?i)|

||||||` "\n" $s -}} {{- $s = replaceRE `(?i)` " | " $s -}} {{- $s = replaceRE `<[^>]+>` "" $s -}} {{- $s = htmlUnescape $s -}} {{- $s = replaceRE `[ \t\x{00a0}]+` " " $s -}} {{- $s = replaceRE `(?m)^ +| +$` "" $s -}} {{- $s = replaceRE `\*\*\s*\*\*` "" $s -}} {{- $s = replaceRE `(?m)^- *$\n?` "" $s -}} {{- $s = replaceRE `\n{3,}` "\n\n" $s -}} {{- return strings.TrimSpace $s -}}
